```bash
# Step 3: Create node directories and config files
for i in $(seq 0 $((NUM_VALIDATORS - 1))); do
    local dir p2p_port grpc_port
    dir="$(node_dir "$i")"
    p2p_port=$((P2P_BASE + i))
    grpc_port=$((GRPC_BASE + i))

    mkdir -p "${dir}/data"

    # Read this node's ed25519 public key from key material
    local this_ed25519_pub
    this_ed25519_pub=$(python3 -c "import json,sys; d=json.load(open(sys.argv[1])); print(d['ed25519_public_hex'])" "$(key_file "$i")")

    # Write TOML config with [[peers]] table format
    cat > "${dir}/config.toml" <<TOML
# slay3rd node ${i} config — auto-generated by scripts/testnet.sh
validator_index = ${i}
chain_id = "slay3r-testnet-1"
p2p_listen = "127.0.0.1:${p2p_port}"
grpc_listen = "127.0.0.1:${grpc_port}"
data_dir = "${dir}/data"
bls_key_path = "$(key_file "$i")"
identity_key_path = "$(key_file "$i")"
genesis_path = "${TESTNET_DIR}/genesis.json"
mempool_max_pending = 10000
leader_timeout_ms = 3000
certification_timeout_ms = 5000
TOML

    # Add [[peers]] entries for all OTHER validators
    for j in $(seq 0 $((NUM_VALIDATORS - 1))); do
        if [[ "$j" != "$i" ]]; then
            local peer_p2p_port peer_ed25519_pub
            peer_p2p_port=$((P2P_BASE + j))
            peer_ed25519_pub=$(python3 -c "import json,sys; d=json.load(open(sys.argv[1])); print(d['ed25519_public_hex'])" "$(key_file "$j")")

            cat >> "${dir}/config.toml" <<TOML

[[peers]]
address = "127.0.0.1:${peer_p2p_port}"
public_key = "${peer_ed25519_pub}"
TOML
        fi
    done

    log "  Node ${i}: config at ${dir}/config.toml (P2P=:${p2p_port}, gRPC=:${grpc_port})"
done
```

**Create `scripts/docker-compose.yml`:**

```yaml
# docker-compose.yml — 3-node Layer testnet (multi-host topology)
#
# This demonstrates running slay3rd nodes as separate containers with
# real network isolation. For localhost testing, use scripts/testnet.sh.
#
# Prerequisites:
#   1. Build: docker build -t slay3rd -f Dockerfile .
#   2. Generate keys: scripts/testnet.sh start (to create key material)
#   3. Run: docker compose -f scripts/docker-compose.yml up
#
# Each node has its own data volume and config. The P2P addresses use
# Docker service names (node0, node1, node2) resolved via Docker DNS.

services:
  node0:
    image: slay3rd:latest
    container_name: slay3r-node0
    command: ["/app/slay3rd", "/config/node0.toml"]
    ports:
      - "7001:7001"
      - "9090:9090"
    volumes:
      - node0-data:/data
      - ./testnet-config:/config:ro
      - ./testnet-keys:/keys:ro
    environment:
      - RUST_LOG=info
    networks:
      - slay3r-net

  node1:
    image: slay3rd:latest
    container_name: slay3r-node1
    command: ["/app/slay3rd", "/config/node1.toml"]
    ports:
      - "7002:7002"
      - "9091:9091"
    volumes:
      - node1-data:/data
      - ./testnet-config:/config:ro
      - ./testnet-keys:/keys:ro
    environment:
      - RUST_LOG=info
    networks:
      - slay3r-net

  node2:
    image: slay3rd:latest
    container_name: slay3r-node2
    command: ["/app/slay3rd", "/config/node2.toml"]
    ports:
      - "7003:7003"
      - "9092:9092"
    volumes:
      - node2-data:/data
      - ./testnet-config:/config:ro
      - ./testnet-keys:/keys:ro
    environment:
      - RUST_LOG=info
    networks:
      - slay3r-net

volumes:
  node0-data:
  node1-data:
  node2-data:

networks:
  slay3r-net:
    driver: bridge
```

Note: Docker Compose configs would use `node0:7001`, `node1:7002`, `node2:7003` as peer addresses (Docker DNS). The file documents the multi-host model per CONTEXT.md.
